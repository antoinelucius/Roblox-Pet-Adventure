local function colorFromData(c)
    if typeof(c) == "Color3" then return c end
    if typeof(c) == "table" then return Color3.fromRGB(c[1] or 255, c[2] or 255, c[3] or 255) end
    return Color3.fromRGB(150, 150, 150)
end

local function makeCorner(parent, radius)

local function fmtInt(n)
    return ((tostring(math.floor(n or 0)):reverse():gsub("(%d%d%d)", "%1 "):reverse():gsub("^%s+", "")))
end

local Anim = {}
Anim.FAST = TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
Anim.SOFT = TweenInfo.new(0.28, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
Anim.BOUNCY = TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
Anim.live = setmetatable({}, { __mode = "k" })
Anim.home = setmetatable({}, { __mode = "k" })
Anim.token = setmetatable({}, { __mode = "k" })
Anim.opened = {}

function Anim.info(time, style, dir, delay, repeatCount, reverses)
    return TweenInfo.new(time, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out, repeatCount or 0, reverses or false, delay or 0)
end

-- joue un tween en annulant le precedent sur le meme element (evite qu'ils se battent)
function Anim.play(inst, info, props, key)
    key = key or "main"
    local slot = Anim.live[inst]
    if not slot then
        slot = {}
        Anim.live[inst] = slot
    end
    if slot[key] then slot[key]:Cancel() end
    local tw = TweenService:Create(inst, info, props)
    slot[key] = tw
    tw:Play()
    return tw
end

-- UIScale dedie aux animations de zoom
function Anim.scale(guiObject)
    local s = guiObject:FindFirstChild("AnimScale")
    if not s then
        s = Instance.new("UIScale")
        s.Name = "AnimScale"
        s.Parent = guiObject
    end
    return s
end

-- place le point d'ancrage au centre sans deplacer l'element (pour zoomer depuis le centre)
function Anim.center(guiObject)
    local a = guiObject.AnchorPoint
    if a.X == 0.5 and a.Y == 0.5 then return guiObject end
    local s, p = guiObject.Size, guiObject.Position
    guiObject.AnchorPoint = Vector2.new(0.5, 0.5)
    guiObject.Position = UDim2.new(
        p.X.Scale + s.X.Scale * (0.5 - a.X), p.X.Offset + s.X.Offset * (0.5 - a.X),
        p.Y.Scale + s.Y.Scale * (0.5 - a.Y), p.Y.Offset + s.Y.Offset * (0.5 - a.Y))
    return guiObject
end

-- petit "pop" : l'element apparait a l'echelle from puis revient a 1 avec un rebond
function Anim.pop(guiObject, from, time, delay)
    local s = Anim.scale(guiObject)
    s.Scale = from or 1.25
    Anim.play(s, Anim.info(time or 0.3, Enum.EasingStyle.Back, nil, delay), { Scale = 1 }, "scale")
end

-- survol / appui : le bouton grossit, s'ecrase puis rebondit
function Anim.button(btn)
    if btn:GetAttribute("AnimReady") then return btn end
    btn:SetAttribute("AnimReady", true)
    local s = Anim.scale(btn)
    btn.MouseEnter:Connect(function() Anim.play(s, Anim.FAST, { Scale = 1.06 }, "scale") end)
    btn.MouseLeave:Connect(function() Anim.play(s, Anim.FAST, { Scale = 1 }, "scale") end)
    btn.MouseButton1Down:Connect(function() Anim.play(s, Anim.FAST, { Scale = 0.9 }, "scale") end)
    btn.MouseButton1Up:Connect(function() Anim.play(s, Anim.BOUNCY, { Scale = 1 }, "scale") end)
    return btn
end

-- ouverture / fermeture animee d'un panneau
function Anim.show(p)
    Anim.opened[p] = true
    p.Visible = true
    local s = Anim.scale(p)
    s.Scale = 0.75
    Anim.play(s, Anim.BOUNCY, { Scale = 1 }, "scale")
end

function Anim.hide(p)
    if not Anim.opened[p] then return end
    Anim.opened[p] = nil
    local tw = Anim.play(Anim.scale(p), Anim.info(0.14, Enum.EasingStyle.Quad, Enum.EasingDirection.In), { Scale = 0.75 }, "scale")
    tw.Completed:Connect(function(playbackState)
        if playbackState == Enum.PlaybackState.Completed and not Anim.opened[p] then
            p.Visible = false
        end
    end)
end

-- renvoie true si le panneau vient d'etre ouvert
function Anim.toggle(p)
    if Anim.opened[p] then
        Anim.hide(p)
        return false
    end
    Anim.show(p)
    return true
end

-- secousse qui s'attenue (l'element doit avoir une position fixe)
function Anim.shake(guiObject, power, time)
    local home = Anim.home[guiObject]
    if not home then
        home = guiObject.Position
        Anim.home[guiObject] = home
    end
    local id = (Anim.token[guiObject] or 0) + 1
    Anim.token[guiObject] = id
    task.spawn(function()
        local t0 = os.clock()
        while os.clock() - t0 < time and Anim.token[guiObject] == id and guiObject.Parent do
            local k = 1 - (os.clock() - t0) / time
            guiObject.Position = home + UDim2.fromOffset((math.random() - 0.5) * 2 * power * k, (math.random() - 0.5) * 2 * power * k)
            RunService.RenderStepped:Wait()
        end
        if Anim.token[guiObject] == id then guiObject.Position = home end
    end)
end

-- contour (stroke) autour d'un cadre, ou autour du texte si onText = true
function Anim.stroke(parent, color, thickness, transparency, onText)
    local s = Instance.new("UIStroke")
    s.Color = color
    s.Thickness = thickness or 2
    s.Transparency = transparency or 0
    s.ApplyStrokeMode = onText and Enum.ApplyStrokeMode.Contextual or Enum.ApplyStrokeMode.Border
    s.Parent = parent
    return s
end

-- texte qui jaillit, monte puis s'efface (degats, gains de coins...)
function Anim.floatText(parent, text, color, pos, size, z)
    local l = Instance.new("TextLabel")
    l.AnchorPoint = Vector2.new(0.5, 0.5)
    l.Position = pos
    l.Size = UDim2.fromOffset(260, size + 12)
    l.BackgroundTransparency = 1
    l.Text = text
    l.TextColor3 = color
    l.Font = Enum.Font.GothamBlack
    l.TextSize = size
    l.ZIndex = z or 30
    local st = Anim.stroke(l, Color3.fromRGB(0, 0, 0), 2, 0, true)
    l.Parent = parent
    Anim.pop(l, 0.3, 0.25)
    TweenService:Create(l, Anim.info(0.95), { Position = pos + UDim2.fromOffset(math.random(-24, 24), -64) }):Play()
    TweenService:Create(l, Anim.info(0.35, Enum.EasingStyle.Linear, nil, 0.6), { TextTransparency = 1 }):Play()
    TweenService:Create(st, Anim.info(0.35, Enum.EasingStyle.Linear, nil, 0.6), { Transparency = 1 }):Play()
    task.delay(1.05, function() l:Destroy() end)
    return l
end

-- compteur anime : le nombre defile de from a to
function Anim.count(label, from, to, time, format)
    local id = (Anim.token[label] or 0) + 1
    Anim.token[label] = id
    if from == to or time <= 0 then
        label.Text = format(to)
        return
    end
    task.spawn(function()
        local t0 = os.clock()
        while os.clock() - t0 < time and Anim.token[label] == id and label.Parent do
            local k = (os.clock() - t0) / time
            k = 1 - (1 - k) * (1 - k)
            label.Text = format(from + (to - from) * k)
            RunService.RenderStepped:Wait()
        end
        if Anim.token[label] == id then label.Text = format(to) end
    end)
end

-- rayons lumineux tournants (victoire, eclosion, renaissance)
function Anim.rays(parent, position, z)
    local holder = Instance.new("Frame")
    holder.Name = "Rays"
    holder.AnchorPoint = Vector2.new(0.5, 0.5)
    holder.Position = position
    holder.Size = UDim2.fromOffset(10, 10)
    holder.BackgroundTransparency = 1
    holder.Visible = false
    holder.ZIndex = z
    for i = 1, 8 do
        local ray = Instance.new("Frame")
        ray.AnchorPoint = Vector2.new(0.5, 0.5)
        ray.Position = UDim2.fromScale(0.5, 0.5)
        ray.Size = UDim2.fromOffset(900, 34)
        ray.Rotation = (i - 1) * 22.5
        ray.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
        ray.BackgroundTransparency = 0.55
        ray.BorderSizePixel = 0
        ray.ZIndex = z
        local g = Instance.new("UIGradient")
        g.Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 1), NumberSequenceKeypoint.new(0.5, 0), NumberSequenceKeypoint.new(1, 1)
        })
        g.Parent = ray
        ray.Parent = holder
    end
    holder.Parent = parent
    TweenService:Create(holder, TweenInfo.new(9, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut, -1), { Rotation = 360 }):Play()
    return holder
end

function Anim.setRays(holder, color, visible)
    holder.Visible = visible
    if not visible then return end
    for _, ray in ipairs(holder:GetChildren()) do
        if ray:IsA("Frame") then ray.BackgroundColor3 = color end
    end
    Anim.pop(holder, 0.1, 0.5)
end

-- ---------- camera de combat (vue cinematique de l'arene) ----------

local Cam = { active = false, shake = 0, punch = 0, focus = 0, focusGoal = 0 }

function Cam.stop()
    if not Cam.active then return end
    Cam.active = false
    pcall(function() RunService:UnbindFromRenderStep("MWS_BattleCam") end)
    local camera = workspace.CurrentCamera
    if camera then
        camera.CameraType = Enum.CameraType.Custom
        camera.FieldOfView = Cam.prevFov or 70
        local char = player.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then camera.CameraSubject = hum end
    end
end

-- rend la camera au joueur des qu'il a quitte l'arene (ou au plus tard apres maxWait secondes)
function Cam.release(maxWait)
    if Cam.active then Cam.releaseAt = os.clock() + (maxWait or 0) end
end

-- secousse (shake, en studs) + coup de zoom (punch, 0 a 1)
function Cam.kick(shake, punch)
    if not Cam.active then return end
    Cam.shake = math.max(Cam.shake, shake)
    Cam.punch = math.max(Cam.punch, punch or 0)
end

function Cam.update(dt)
    local camera = workspace.CurrentCamera
    if not Cam.active or not camera then return end
    local now = os.clock()
    local t = now - Cam.t0
    local origin = Cam.origin

    -- securite : si le joueur n'est plus dans l'arene (combat fini, mort, retour), on rend la camera
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local away = (not hrp) or (hrp.Position - origin).Magnitude > 75
    if (away and t > 3) or (Cam.releaseAt and (away or now > Cam.releaseAt)) then
        Cam.stop()
        BUI.closeBattle()
        return
    end

    -- introduction : la camera part de derriere le joueur et tourne jusqu'a la vue de cote
    local k = math.clamp(t / 1.8, 0, 1)
    local e = 1 - (1 - k) * (1 - k) * (1 - k)
    local angle = -(1 - e) * 1.4 + math.sin(t * 0.35) * 0.08
    local dist = 27 + (1 - e) * 13
    local height = 13 + (1 - e) * 8
    Cam.focus = Cam.focus + (Cam.focusGoal - Cam.focus) * math.min(1, dt * 5)
    local target = origin + Vector3.new(Cam.focus, -4.5, 0)
    local eye = origin + Vector3.new(math.sin(angle) * dist + Cam.focus * 0.6, height, math.cos(angle) * dist)

    -- victoire : la camera vient tourner autour du pet du joueur
    if Cam.mode == "win" then
        local u = now - Cam.modeT
        local m = math.min(1, u / 1.2)
        m = 1 - (1 - m) * (1 - m)
        local a2 = 0.5 + u * 0.35
        eye = eye:Lerp(origin + Vector3.new(-10 + math.sin(a2) * 15, 8, math.cos(a2) * 15), m)
        target = target:Lerp(origin + Vector3.new(-10, -2, 0), m)
    end

    Cam.shake = Cam.shake * math.exp(-dt * 8)
    Cam.punch = Cam.punch * math.exp(-dt * 5)
    local sh = Cam.shake
    local jitter = Vector3.new((math.random() - 0.5) * sh, (math.random() - 0.5) * sh, (math.random() - 0.5) * sh)
    camera.CFrame = CFrame.lookAt(eye + jitter, target + jitter * 0.3)
    camera.FieldOfView = 58 - Cam.punch * 9
end

function Cam.start(origin, skipIntro)
    if not BATTLE_CAMERA or typeof(origin) ~= "Vector3" then return end
    local camera = workspace.CurrentCamera
    if not camera then return end
    Cam.origin = origin
    Cam.releaseAt = nil
    Cam.mode = nil
    if Cam.active and skipIntro then return end
    Cam.t0 = os.clock() - (skipIntro and 5 or 0)
    Cam.shake, Cam.punch, Cam.focus, Cam.focusGoal = 0, 0, 0, 0
    if Cam.active then return end
    Cam.active = true
    Cam.prevFov = camera.FieldOfView
    camera.CameraType = Enum.CameraType.Scriptable
    RunService:BindToRenderStep("MWS_BattleCam", Enum.RenderPriority.Camera.Value + 1, Cam.update)
end

-- journal du combat : le texte s'ecrit lettre par lettre
function BUI.setLog(text)
    text = text or ""
    if text == Battle.logText then return end
    Battle.logText = text
    Battle.logId = (Battle.logId or 0) + 1
    local id = Battle.logId
    local n = utf8.len(text) or #text
    local dur = math.clamp(n * 0.012, 0.15, 0.6)
    battleLog.Text = text
    battleLog.MaxVisibleGraphemes = 0
    task.spawn(function()
        local t0 = os.clock()
        while Battle.logId == id and os.clock() - t0 < dur do
            battleLog.MaxVisibleGraphemes = math.floor(n * (os.clock() - t0) / dur)
            RunService.RenderStepped:Wait()
        end
        if Battle.logId == id then battleLog.MaxVisibleGraphemes = -1 end
    end)
end

local function hpColor(side, pct)
    if side == "player" then
        return Color3.fromRGB(math.floor(50 + (1 - pct) * 150), math.floor(200 * pct), 50)
    end
    return Color3.fromRGB(math.floor(200 * pct + 50), math.floor(80 * (1 - pct)), math.floor(50 * (1 - pct)))
end

-- barre de vie : descend en douceur, suivie de la trainee blanche ; le nombre de PV defile
function BUI.setHp(side, pet, idx)
    local bar, ghost, text = BUI.bar[side], BUI.ghost[side], BUI.hpText[side]
    local pct = math.clamp(pet.hp / math.max(1, pet.maxHp), 0, 1)
    local size = UDim2.new(pct, 0, 1, 0)
    local maxHp = pet.maxHp
    local function format(v)
        return math.floor(v + 0.5) .. " / " .. maxHp .. " HP"
    end
    local prev = Battle.shown[side]
    if not prev or prev.idx ~= idx then
        -- nouveau pet : valeurs posees d'un coup, l'icone surgit
        Anim.play(bar, Anim.info(0.05), { Size = size, BackgroundColor3 = hpColor(side, pct) })
        Anim.play(ghost, Anim.info(0.05), { Size = size })
        Anim.count(text, pet.hp, pet.hp, 0, format)
        Anim.pop(BUI.icon[side], 0.2, 0.4)
    else
        Anim.play(bar, Anim.info(0.25), { Size = size, BackgroundColor3 = hpColor(side, pct) })
        Anim.play(ghost, Anim.info(0.5, nil, nil, (pct < prev.pct) and 0.3 or 0), { Size = size })
        Anim.count(text, prev.hp, pet.hp, 0.35, format)
    end
    Battle.shown[side] = { idx = idx, pct = pct, hp = pet.hp, maxHp = maxHp }
end

-- K.O. : la barre tombe a zero et l'icone s'ecrase avant l'arrivee du pet suivant
function BUI.drain(side)
    local prev = Battle.shown[side]
    if not prev then return end
    local maxHp = prev.maxHp
    Anim.play(BUI.bar[side], Anim.info(0.25), { Size = UDim2.new(0, 0, 1, 0) })
    Anim.play(BUI.ghost[side], Anim.info(0.5, nil, nil, 0.3), { Size = UDim2.new(0, 0, 1, 0) })
    Anim.count(BUI.hpText[side], prev.hp, 0, 0.35, function(v)
        return math.floor(v + 0.5) .. " / " .. maxHp .. " HP"
    end)
    prev.hp, prev.pct = 0, 0
    Anim.play(Anim.scale(BUI.icon[side]), Anim.info(0.3, Enum.EasingStyle.Back, Enum.EasingDirection.In, 0.25), { Scale = 0.05 }, "scale")
end

-- impact d'une attaque : secousse du cadre, flash, degats flottants, camera
function BUI.impact(target, fx)
    local frame = BUI.frame[target]
    local special = STYLE_NAMES[fx.style] ~= nil
    local heavy = special or fx.crit
    Anim.shake(frame, heavy and 16 or 8, heavy and 0.45 or 0.3)
    local st = BUI.frameStroke[target]
    st.Color = WHITE
    st.Transparency = 0
    Anim.play(st, Anim.info(0.5), { Transparency = 1 })
    if fx.damage then
        Anim.floatText(frame, "-" .. fmtInt(fx.damage),
            (target == "npc") and Color3.fromRGB(255, 240, 120) or Color3.fromRGB(255, 90, 90),
            UDim2.fromScale(0.5, 0.35), heavy and 36 or 26, 30)
    end
    if target == "player" then
        BUI.screenFlash(Color3.fromRGB(255, 40, 40), special and 0.4 or 0.22, 0.45)
    elseif special then
        BUI.screenFlash((typeof(fx.color) == "Color3") and fx.color or WHITE, 0.35, 0.5)
    elseif fx.crit then
        BUI.screenFlash(WHITE, 0.22, 0.3)
    end
    if fx.crit then BUI.announce("COUP CRITIQUE !", Color3.fromRGB(255, 220, 80)) end
    local shake = 0.55
    if fx.style == "DivineStrike" then shake = 2.6 elseif special then shake = 1.8 elseif fx.crit then shake = 1.2 end
    Cam.kick(shake, heavy and 1 or 0.25)
    Cam.focusGoal = 0
end

-- ---------- mise a jour de l'interface ----------

local function updateBattleUI(data)
    battleData = data
    if not data then return end

    local myPet = data.playerPets[data.playerCurrent]
    local enemyPet = data.npcPets[data.npcCurrent]

    if myPet then
        playerPetName.Text = myPet.name
        playerPetIcon.BackgroundColor3 = colorFromData(myPet.color)
        local myImg = getPetImage(myPet.name)
        playerPetImg.Image = myImg or ""
        playerPetImg.Visible = myImg ~= nil
        BUI.setHp("player", myPet, data.playerCurrent)
        playerPetCount.Text = "Pet " .. data.playerCurrent .. " / " .. #data.playerPets
    end

    if enemyPet then
        npcPetName.Text = enemyPet.name
        npcPetIcon.BackgroundColor3 = colorFromData(enemyPet.color)
        local enImg = getPetImage(enemyPet.name)
        npcPetImg.Image = enImg or ""
        npcPetImg.Visible = enImg ~= nil
        BUI.setHp("npc", enemyPet, data.npcCurrent)
        npcPetCount.Text = "Pet " .. data.npcCurrent .. " / " .. #data.npcPets
    end

    BUI.setLog(data.log)
    fleeBtn.Visible = (data.active == true)

    local charge = data.playerCharge or 0
    local maxCharge = data.playerMaxCharge or 100
    BUI.setCharge(maxCharge > 0 and (charge / maxCharge) or 0)
    superBtn.Visible = false

    if data.active then
        if data.turn == "player" then
            BUI.setWaiting(false)
            BUI.showActions(data)
        else
            BUI.hideActions()
            BUI.setWaiting(true)
        end
    else
        BUI.hideActions()
        BUI.setWaiting(false)
        if Battle.ended ~= battleSerial then
            Battle.ended = battleSerial
            local fx = data.fx
            local result = fx and fx.result
            if not result then
                if string.find(data.log or "", "VICTOIRE") then
                    result = "win"
                elseif string.find(data.log or "", "Defaite") then
                    result = "lose"
                end
            end
            if result then BUI.showResult(result, fx) end
            local serial = battleSerial
            task.delay((result == "flee") and 1.3 or 3.6, function()
                if serial ~= battleSerial then return end
                BUI.closeBattle()
            end)
        end
    end
end

-- applique une mise a jour au moment de l'impact (l'interface reste synchronisee avec l'arene)
function BUI.apply(data, seq)
    if seq < Battle.applied then return end
    Battle.applied = seq
    local fx = data.fx
    if fx and fx.kind == "strike" then
        local target = (fx.actor == "player") and "npc" or "player"
        BUI.impact(target, fx)
        local prev = Battle.shown[target]
        local cur = (target == "npc") and data.npcCurrent or data.playerCurrent
        if fx.ko and prev and prev.idx ~= cur then
            -- K.O. : la barre tombe a zero, puis le pet suivant entre en scene
            BUI.drain(target)
            BUI.setLog(data.log)
            local serial = battleSerial
            task.delay(0.9, function()
                if serial ~= battleSerial or Battle.applied ~= seq then return end
                updateBattleUI(data)
            end)
            return
        end
    end
    updateBattleUI(data)
end

-- reception d'une mise a jour du serveur : annonce l'attaque tout de suite, applique le reste a l'impact
local function onBattleData(data)
    if type(data) ~= "table" then return end
    Battle.seq = Battle.seq + 1
    local seq = Battle.seq
    local fx = (type(data.fx) == "table") and data.fx or nil
    if fx and fx.kind == "strike" then
        BUI.hideActions()
        local name = STYLE_NAMES[fx.style]
        if name then BUI.announce(name, (typeof(fx.color) == "Color3") and fx.color or WHITE) end
        Cam.focusGoal = (fx.actor == "player") and 3 or -3
        local st = BUI.frameStroke[fx.actor]
        if st then
            st.Color = BUI.sideColor[fx.actor]
            st.Transparency = 0
            Anim.play(st, Anim.info(0.6), { Transparency = 1 })
        end
    elseif fx and fx.kind == "switch" then
        BUI.hideActions()
    end
    local wait = (fx and tonumber(fx.impact)) or 0
    if wait > 0 then
        local serial = battleSerial
        task.delay(wait, function()
            if serial ~= battleSerial then return end
            BUI.apply(data, seq)
        end)
    else
        BUI.apply(data, seq)
    end
end

local HATCH_TIER = { Legendaire = 2, Mythique = 2, Divin = 3, Secret = 3 }

-- eclosion : l'oeuf tremble de plus en plus fort, flash, puis l'animal surgit devant les rayons
function Overlay.hatch(pet)
    local id = Overlay.begin(0.45)
    local rc = RARITY_COLORS[pet.rarity] or WHITE
    local tier = HATCH_TIER[pet.rarity] or 1
    hatchLabel.TextSize = 30
    hatchLabel.TextColor3 = WHITE
    hatchLabel.Text = "Eclosion..."
    Anim.pop(hatchLabel, 0.6, 0.3)
    local egg = Overlay.egg
    local eggScale = Anim.scale(egg)
    egg.Rotation = 0
    eggScale.Scale = 0.05
    egg.Visible = true
    task.spawn(function()
        local t0 = os.clock()
        local dur = 1.1
        while os.clock() - t0 < dur do
            if id ~= Overlay.id then return end
            local k = (os.clock() - t0) / dur
            egg.Rotation = math.sin(k * k * 45) * (5 + 24 * k)
            eggScale.Scale = math.min(1, 0.05 + k * 6) * (1 + 0.3 * k)
            RunService.RenderStepped:Wait()
        end
        if id ~= Overlay.id then return end
        egg.Visible = false
        BUI.screenFlash(rc, 0.5 + tier * 0.12, 0.55)
        Overlay.rays.Position = UDim2.new(0.5, 0, 0.5, -190)
        Anim.setRays(Overlay.rays, rc, true)
        local img = getPetImage(pet.name)
        hatchImage.Image = img or ""
        hatchImage.BackgroundColor3 = colorFromData(pet.color)
        hatchImage.BackgroundTransparency = img and 1 or 0
        Overlay.imgStroke.Color = rc
        Overlay.imgStroke.Enabled = img == nil
        hatchImage.Visible = true
        Anim.pop(hatchImage, 0.15, 0.45)
        hatchLabel.Text = pet.name .. "\n" .. pet.rarity .. "\n+" .. tostring(pet.income or 0) .. " coins/sec"
        hatchLabel.TextColor3 = rc
        Anim.pop(hatchLabel, 0.4, 0.4)
        if tier >= 2 then Anim.shake(hatchLabel, 5 * tier, 0.4) end
        task.wait(1.6 + tier * 0.3)
        Overlay.finish(id)
    end)
end

local Coin = {}

local function formatCoins(v)
    return fmtInt(v) .. " Coins"
end

StateUpdate.OnClientEvent:Connect(function(newState, income)
    state = newState
    local newIncome = income or 0
    local coins = math.floor(state.coins or 0)
    local prev = Coin.shown
    Coin.shown = coins
    if prev == nil then
        coinLabel.Text = formatCoins(coins)
    else
        -- le compteur defile ; les gros gains (cristaux, combats) et les depenses flottent a cote
        Anim.count(coinLabel, prev, coins, 0.45, formatCoins)
        local delta = coins - prev
        if delta >= math.max(50, incomePerSec * 3) then
            Anim.pop(coinIcon, 1.5, 0.35)
            Anim.floatText(gui, "+" .. fmtInt(delta), Color3.fromRGB(255, 230, 110), UDim2.fromOffset(200, 100), 22, 30)
        elseif delta < 0 then
            Anim.floatText(gui, "-" .. fmtInt(-delta), Color3.fromRGB(255, 110, 110), UDim2.fromOffset(200, 100), 20, 30)
            Anim.shake(topBar, 4, 0.25)
        end
    end
    if Coin.income ~= nil and Coin.income ~= newIncome then Anim.pop(incomeLabel, 1.25, 0.3) end
    Coin.income = newIncome
    incomePerSec = newIncome
    incomeLabel.Text = "+" .. incomePerSec .. " /sec"
    local level = state.rebirthLevel or 0
    if Coin.rebirth ~= nil and Coin.rebirth ~= level then Anim.pop(rebirthLabel, 1.6, 0.45) end
    Coin.rebirth = level
    rebirthLabel.Text = "Renaissance: " .. tostring(level)
    if panel.Visible then refreshPanel() end
    if combatPanel.Visible then refreshCombatPanel() end
end)

                buy.BackgroundColor3 = Color3.fromRGB(80, 80, 95)
            else
                buy.Text = "R$ " .. tostring(item.price or "?")
                buy.BackgroundColor3 = Color3.fromRGB(0, 150, 90)
            end
            buy.Parent = card
            buy.MouseButton1Click:Connect(function()
                if not item.owned then ShopBuy:FireServer(item.key) end
            end)
            Anim.button(Anim.center(buy))
        end
    end

    ShopInfo.OnClientEvent:Connect(function(list, meta)
        shopItems = list or {}
        shopMeta = meta or {}
        if shopMeta.now then clockOffset = shopMeta.now - os.time() end
        if shopPanel.Visible then refreshShop() end
    end)

    task.spawn(function()
        while true do
            task.wait(1)
            if shopPanel.Visible then
                local now = os.time() + clockOffset
                local parts = {}
                if (shopMeta.speedUntil or 0) > now then
                    table.insert(parts, "\u{26A1} Vitesse " .. fmtTime(shopMeta.speedUntil - now))
                end
                if (shopMeta.luckUntil or 0) > now then
                    table.insert(parts, "\u{1F340} Chance " .. fmtTime(shopMeta.luckUntil - now))
                end
                shopStatus.Text = (#parts > 0) and ("Bonus actifs : " .. table.concat(parts, "   |   ")) or "Aucun bonus actif"
            end
        end
    end)

    -- ---------- mondes ----------
    local function refreshWorlds(animate)
        for _, c in ipairs(worldScroll:GetChildren()) do
            if c:IsA("Frame") then c:Destroy() end
        end
        for i, w in ipairs(worldList) do
            local row = Instance.new("Frame")
            row.LayoutOrder = i
            row.Size = UDim2.new(1, -8, 0, 80)
            row.BackgroundColor3 = w.unlocked and Color3.fromRGB(35, 55, 45) or Color3.fromRGB(55, 40, 40)
            row.ZIndex = 6
            makeCorner(row, 12)
            row.Parent = worldScroll
            if animate then Anim.pop(row, 0.7, 0.3, i * 0.06) end

            local nameL = Instance.new("TextLabel")
            nameL.Size = UDim2.new(1, -150, 0, 30)
            nameL.Position = UDim2.new(0, 14, 0, 10)
            nameL.BackgroundTransparency = 1
            nameL.Text = "Monde " .. i .. " - " .. (w.name or "")
            nameL.TextColor3 = WHITE
            nameL.Font = Enum.Font.GothamBold
            nameL.TextSize = 17
            nameL.TextXAlignment = Enum.TextXAlignment.Left
            nameL.ZIndex = 7
            nameL.Parent = row

            local infoL = Instance.new("TextLabel")
            infoL.Size = UDim2.new(1, -150, 0, 32)
            infoL.Position = UDim2.new(0, 14, 0, 42)
            infoL.BackgroundTransparency = 1
            local needRb = w.rebirth or 0
            local myRb = state.rebirthLevel or 0
            local infoText = "Debloque"
            if not w.unlocked then
                infoText = "Prix : " .. fmtNum(w.unlock) .. " coins (une fois)"
                if needRb > 0 then
                    infoText = infoText .. "  +  " .. needRb .. " renaissance(s) (tu en as " .. myRb .. ")"
                end
            end
            infoL.Text = infoText
            infoL.TextColor3 = w.unlocked and Color3.fromRGB(120, 255, 160) or Color3.fromRGB(255, 200, 120)
            infoL.Font = Enum.Font.Gotham
            infoL.TextSize = 12
            infoL.TextWrapped = true
            infoL.TextYAlignment = Enum.TextYAlignment.Top
            infoL.TextXAlignment = Enum.TextXAlignment.Left
            infoL.ZIndex = 7
            infoL.Parent = row

            local go = Instance.new("TextButton")
            go.Size = UDim2.new(0, 120, 0, 40)
            go.Position = UDim2.new(1, -132, 0.5, -20)
            go.BackgroundColor3 = w.unlocked and Color3.fromRGB(50, 150, 90) or Color3.fromRGB(200, 140, 40)
            go.Text = w.unlocked and "Y aller" or "Debloquer"
            go.TextColor3 = WHITE
            go.Font = Enum.Font.GothamBold
            go.TextSize = 15
            go.ZIndex = 7
            makeCorner(go, 8)
            go.Parent = row
            go.MouseButton1Click:Connect(function()
                WorldTeleport:FireServer(i)
                Anim.hide(worldPanel)
            end)
            Anim.button(Anim.center(go))
        end
    end

        worldList = list or {}
